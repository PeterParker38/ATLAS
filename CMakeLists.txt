cmake_minimum_required(VERSION 3.22)
project(Atlas VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Threads REQUIRED)
find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
set(ATLAS_SANITIZER "" CACHE STRING "Sanitizer: address or thread")
add_library(atlas_options INTERFACE)
target_compile_options(atlas_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ATLAS_SANITIZER STREQUAL "address")
  target_compile_options(atlas_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(atlas_options INTERFACE -fsanitize=address,undefined -no-pie)
elseif(ATLAS_SANITIZER STREQUAL "thread")
  target_compile_options(atlas_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(atlas_options INTERFACE -fsanitize=thread)
elseif(NOT ATLAS_SANITIZER STREQUAL "")
  message(FATAL_ERROR "ATLAS_SANITIZER must be empty, address, or thread")
endif()
add_library(atlas_protocol src/protocol.cpp)
target_include_directories(atlas_protocol PUBLIC include)
target_link_libraries(atlas_protocol PUBLIC atlas_options)
add_library(atlas_memory_storage src/memory_storage.cpp)
target_link_libraries(atlas_memory_storage PUBLIC atlas_protocol)
add_library(atlas_transport src/server.cpp src/client.cpp)
target_include_directories(atlas_transport SYSTEM PUBLIC ${ASIO_INCLUDE_DIR})
target_compile_definitions(atlas_transport PUBLIC ASIO_STANDALONE ASIO_NO_DEPRECATED)
target_link_libraries(atlas_transport PUBLIC atlas_protocol Threads::Threads)
add_executable(atlas-node apps/node.cpp)
target_link_libraries(atlas-node PRIVATE atlas_transport atlas_memory_storage)
add_executable(atlas-cli apps/cli.cpp)
target_link_libraries(atlas-cli PRIVATE atlas_transport)
install(TARGETS atlas-node atlas-cli RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(atlas-unit-tests tests/unit.cpp)
  target_link_libraries(atlas-unit-tests PRIVATE atlas_protocol atlas_memory_storage)
  add_test(NAME unit COMMAND atlas-unit-tests)
  add_executable(atlas-transport-tests tests/transport.cpp)
  target_link_libraries(atlas-transport-tests PRIVATE atlas_transport atlas_memory_storage)
  add_test(NAME transport COMMAND atlas-transport-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:atlas-node> $<TARGET_FILE:atlas-cli>)
  set_tests_properties(unit transport integration PROPERTIES TIMEOUT 60)
endif()
